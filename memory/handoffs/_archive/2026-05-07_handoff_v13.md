## 11. B-6 SWA baseline 代码模板

```python
"""B-6 SWA / model-soup baseline.

Compare weight-space averaging (SWA, Izmailov 2018) vs HAWE action-space
averaging. Same w_R21=0.98 sweet spot, plus full sweep.

Zero retraining cost — just load .pt, average parameters, save, eval.
"""
import torch
from pathlib import Path

R21_DIR = Path("results/v4_h50_s49")
WS8_DIR = Path("results/v4_8_warmstart_R21_s49")
OUT_DIR = Path("results/v4_swa_w9802")
OUT_DIR.mkdir(exist_ok=True)

W_R21 = 0.98

for agent_i in range(4):
    sd_r21 = torch.load(R21_DIR / f"agent_{agent_i}_best.pt", map_location="cpu")
    sd_ws8 = torch.load(WS8_DIR / f"agent_{agent_i}_best.pt", map_location="cpu")
    
    # Weight-space average — same keys, same shape
    sd_swa = {k: W_R21 * sd_r21[k] + (1 - W_R21) * sd_ws8[k] 
              for k in sd_r21.keys()}
    
    torch.save(sd_swa, OUT_DIR / f"agent_{agent_i}_best.pt")

print(f"SWA actors saved at {OUT_DIR}")
```
